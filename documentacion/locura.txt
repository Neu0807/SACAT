            'es_extraordinaria': document.querySelector('[name="es_extraordinaria"]')
        };

        // --- [2. FUNCIONES DE ESTADO Y ALERTAS DE GUÍA] ---
        function setValid(el) {
            if (!el) return;
            el.classList.remove('is-invalid');
            el.classList.add('is-valid');
            const fb = el.nextElementSibling;
            if (fb && fb.classList.contains('feedback-box')) fb.innerText = "";
        }

        function setInvalid(el, msg) {
            if (!el) return;
            el.classList.remove('is-valid');
            el.classList.add('is-invalid');
            const fb = el.nextElementSibling;
            if (fb && fb.classList.contains('feedback-box')) fb.innerText = msg;
        }

        // --- [3. REGLAS DE NEGOCIO (Edad mínima 16 años)] ---
        const hoy = new Date();
        const maxFechaNac = new Date(hoy.getFullYear() - 16, hoy.getMonth(), hoy.getDate());
        if (inpNacimiento) inpNacimiento.max = maxFechaNac.toISOString().split("T")[0];

        // --- [4. MÁSCARA MM/AAAA PARA VENCIMIENTO] ---
        if (inpVencimiento) {
            inpVencimiento.addEventListener('input', function() {
                let val = this.value.replace(/\D/g, ''); 
                if (val.length > 2) this.value = val.substring(0, 2) + '/' + val.substring(2, 6);
                else this.value = val;
                if (val.length === 6) setValid(this);
            });
        }

        // --- [5. RESETEO AL CAMBIAR TIPO DE DOCUMENTO] ---
        tipoDoc.addEventListener('change', function() {
            inpCedula.readOnly = false;
            inpCedula.disabled = false;
            inpCedula.value = '';
            inpCedula.style.backgroundColor = '';
            const sigla = this.options[this.selectedIndex].getAttribute('data-sigla');
            inpCedula.placeholder = (sigla === 'P') ? "Alfanumérico (Pasaporte)" : "Solo números (Mín 6)";
            
            // RESET DE ARCHIVO: Vuelve a ser obligatorio para registros nuevos
            inpArchivo.required = true;
            if(feedbackArchivo) feedbackArchivo.innerHTML = '';
            inpArchivo.classList.remove('is-valid', 'is-invalid');
            
            document.querySelectorAll('.is-valid, .is-invalid').forEach(el => {
                if(el.id !== 'archivo_cedula') el.classList.remove('is-valid', 'is-invalid');
            });
        });

        // --- [6. BÚSQUEDA LOCAL E INTELIGENCIA DE ARCHIVO ACTUALIZACIÓN] ---
        inpCedula.addEventListener('change', function() {
            const val = this.value.trim();
            const fbC = document.getElementById('feedback_cedula');
            if (val.length >= 6 && !this.readOnly) {
                if(fbC) fbC.innerText = "Consultando base de datos interna...";
                fetch(`{% url 'verificar_cedula_ajax' %}?cedula=${val}`)
                .then(r => r.json())
                .then(data => {
                    if (data.exists) {
                        const p = data.persona;
                        this.readOnly = true;
                        tipoDoc.style.pointerEvents = 'none';
                        tipoDoc.style.backgroundColor = '#f8f9fa';

                        const autoLlenar = (id, valor) => {
                            const f = fields[id];
                            if (f) { f.value = valor || ''; setValid(f); }
                        };

                        autoLlenar('nombre1', p.nombre1); autoLlenar('nombre2', p.nombre2);
                        autoLlenar('apellido1', p.apellido1); autoLlenar('apellido2', p.apellido2);
                        autoLlenar('lugar_nacimiento', p.lugar_nacimiento);
                        if(inpNacimiento) { inpNacimiento.value = p.fecha_nacimiento || ''; setValid(inpNacimiento); }
                        if(inpVencimiento) { inpVencimiento.value = p.vencimiento || ''; setValid(inpVencimiento); }
                        autoLlenar('sexo', p.sexo); autoLlenar('estado_civil', p.estado_civil);

                        // 📂 LÓGICA DE ARCHIVO DIGITAL:
                        if (p.url_archivo) {
                            inpArchivo.required = false; 
                            setValid(inpArchivo); 
                            if (feedbackArchivo) {
                                feedbackArchivo.innerHTML = `
                                    <div class="alert alert-info py-2 mt-2 mb-0 shadow-sm border-0" style="background-color: #e0f2fe; color: #0369a1;">
                                        <i class="fas fa-file-check me-2"></i>

                                        <small class="text-muted">(Suba uno nuevo solo si desea reemplazarlo)</small>
                                    </div>`;
                            }

                        if (p.es_naturalizado) {
                            checkNat.checked = true;
                            divGaceta.style.display = 'block';
                            autoLlenar('nro_gaceta', p.nro_gaceta);
                            autoLlenar('fecha_gaceta', p.fecha_gaceta);
                            autoLlenar('es_extraordinaria', p.es_extraordinaria);

                    }
                    if(fbC) fbC.innerText = "";
                });
            }

        if (btnSeniat) {
            btnSeniat.onclick = function() {
                const cedula = inpCedula.value.trim();
                if (cedula.length < 6) return Swal.fire('Atención', 'Ingrese una cédula válida para consultar.', 'warning');

                btnSeniat.disabled = true;
                btnSeniat.innerHTML = '<i class="fas fa-brain fa-spin"></i>';

                    html: 'El SACAT está resolviendo la operación matemática y recuperando los nombres oficiales.',

                    didOpen: () => { Swal.showLoading(); }
                });

                fetch(`{% url 'consultar_datos_seniat' %}?cedula=${cedula}`)

                    btnSeniat.disabled = false;
                    btnSeniat.innerHTML = '<i class="fas fa-search"></i>';

                        if (fields.apellido1) { fields.apellido1.value = s.apellido1; setValid(fields.apellido1); }
                        if (fields.apellido2) { fields.apellido2.value = s.apellido2; setValid(fields.apellido2); }
                        if (fields.nombre1) { fields.nombre1.value = s.nombre1; setValid(fields.nombre1); }
                        if (fields.nombre2) { fields.nombre2.value = s.nombre2; setValid(fields.nombre2); }
                        Swal.fire({ icon: 'success', title: 'Sincronizado', text: 'Nombres y apellidos oficiales cargados correctamente.', timer: 2000, showConfirmButton: false });
                    } else {
                        Swal.fire('Atención', data.error, 'error');
                    }
                })
                .catch(() => {
                    Swal.close(); btnSeniat.disabled = false; btnSeniat.innerHTML = '<i class="fas fa-search"></i>';
                    Swal.fire('Error', 'No se pudo contactar con el portal de datos externo.', 'error');
                });
            };
        }

        // --- [8. VALIDACIONES DE ARCHIVO Y NOMBRES] ---
        inpArchivo.addEventListener('change', function() {
            if (this.files.length > 0) {
                if (this.files[0].size > 5 * 1024 * 1024) {
                    setInvalid(this, "El archivo supera los 5MB.");
                    this.value = ""; 
                } else { 
                    setValid(this);
                    if(feedbackArchivo) feedbackArchivo.innerHTML = '<small class="text-success fw-bold">Nuevo archivo listo para subir.</small>';
                }
            }
        });

        form.querySelectorAll('.val-name').forEach(input => {
            input.addEventListener('input', function() { 
                this.value = this.value.toUpperCase();
                if (this.value.trim().length >= 2) setValid(this);
            });
            input.addEventListener('blur', function() {
                if (this.required && !this.value.trim()) setInvalid(this, "Campo obligatorio");
            });
        });

        if (checkNat) {
            checkNat.addEventListener('change', function() {
                divGaceta.style.display = this.checked ? 'block' : 'none';
                divGaceta.querySelectorAll('input, select').forEach(i => {
                    i.required = this.checked;
                    if(!this.checked) i.classList.remove('is-invalid', 'is-valid');
                });
            });
        }

        // --- [9. SUBMIT FINAL (Con Redirección al Paso 2)] ---
        form.addEventListener('submit', function(e) {
            e.preventDefault();
            let esValido = true;
            
            form.querySelectorAll('[required]').forEach(i => {
                if (!i.value.trim() && (i.type !== 'file' || i.required)) {
                    setInvalid(i, "Campo obligatorio");
                    esValido = false;
                }
            });

            if (!esValido) return Swal.fire({ icon: 'warning', title: 'Atención', text: 'Por favor, revise los campos marcados en rojo.' });

            btnSiguiente.disabled = true;
            btnSiguiente.innerHTML = '<span class="spinner-border spinner-border-sm"></span> PROCESANDO...';

            fetch("{% url 'guardar_paso1_v2' %}", {
                method: "POST",
                body: new FormData(form),
                headers: { "X-CSRFToken": "{{ csrf_token }}" }
            })
            .then(res => res.json())
            .then(data => {
                if (data.success) {
                    window.location.href = "{% url 'paso2_contactos_f1' %}";
                } else {
                    Swal.fire({ icon: 'error', title: 'Falla al Guardar', text: data.error });
                    btnSiguiente.disabled = false;
                    btnSiguiente.innerHTML = 'GUARDAR Y CONTINUAR';

                btnSiguiente.disabled = false;
                btnSiguiente.innerHTML = 'GUARDAR Y CONTINUAR';
            });
        });

                this.classList.remove('is-invalid');

            // Regla: Duplicados (Solo si no estamos editando el mismo)
            const existe = Array.from(lista.querySelectorAll('input[name="valor[]"]')).some(i => i.value.toLowerCase() === valor.toLowerCase());
            if (existe) return Swal.fire({ icon: 'error', title: 'Duplicado', text: 'Este dato ya está en la lista.' });

            const cat = tipoDesc.toUpperCase().includes('CORREO') ? 'CORREO' : 'TELÉFONO';

            // Regla: Único Principal
            if (esPrincipal) {
                const actual = Array.from(lista.querySelectorAll('.item-contacto')).find(el => 
                    el.getAttribute('data-cat') === cat && el.querySelector('.check-principal').value === "true"
                );
                if (actual) {
                    const confirm = await Swal.fire({ 
                        title: '¿Sustituir?', 
                        text: `Ya hay un ${cat.toLowerCase()} principal.`, 
                        icon: 'question', 
                        showCancelButton: true, 
                        confirmButtonText: 'Sí, sustituir' 
                    });
                    if (confirm.isConfirmed) {
                        actual.querySelector('.check-principal').value = "false";
                        actual.querySelector('.badge-main').classList.add('d-none');
                        actual.classList.remove('is-main');
                    } else { esPrincipal = false; }
                }
            }

            if(document.getElementById('msgVacio')) document.getElementById('msgVacio').remove();

            const icon = cat === 'CORREO' ? 'fa-envelope' : 'fa-phone-alt';
            const col = document.createElement('div');
            col.className = `col-12 item-contacto ${esPrincipal ? 'is-main' : ''}`;
            col.setAttribute('data-cat', cat);
            
            // 📝 Inyectamos el ID (si existía) para que el backend sepa que es un UPDATE
            const inputId = idContactoEnEdicion ? `<input type="hidden" name="contacto_id[]" value="${idContactoEnEdicion}">` : `<input type="hidden" name="contacto_id[]" value="">`;

            col.innerHTML = `
                <div class="contact-card shadow-sm">
                    <div class="d-flex align-items-center">
                        <div class="icon-box-contact"><i class="fas ${icon}"></i></div>
                        <div>
                            <div class="text-muted fw-bold" style="font-size: 0.65rem;">${tipoDesc}</div>
                            <div class="text-main fw-bold">${valor}</div>
                            <span class="badge bg-success badge-main ${esPrincipal ? '' : 'd-none'}" style="font-size: 0.6rem;">PRINCIPAL</span>
                        </div>
                    </div>
                    <div class="d-flex gap-2">
                        <button type="button" class="btn btn-sm text-primary bg-transparent border-0" 
                                onclick="editarContacto(this, '${idContactoEnEdicion || ''}', '${valor}', '${esPrincipal}')">
                            <i class="fas fa-pencil-alt"></i>
                        </button>
                        <button type="button" class="btn btn-sm text-danger bg-transparent border-0 btn-remove">
                            <i class="fas fa-times fa-lg"></i>
                        </button>
                    </div>
                    ${inputId}
                    <input type="hidden" name="tipo_id[]" value="${tipoId}">
                    <input type="hidden" name="valor[]" value="${valor}">
                    <input type="hidden" name="es_principal[]" class="check-principal" value="${esPrincipal}">
                </div>`;
            
            lista.appendChild(col);

            // Reset de variables y campos
            idContactoEnEdicion = null;
            tempValor.value = ''; 
            tempPrincipal.checked = false; 
            tempValor.classList.remove('is-valid');
        };

        // --- [4. ENVÍO FINAL] ---
        form.addEventListener('submit', function(e) {
            e.preventDefault();
            const items = lista.querySelectorAll('.item-contacto');
            let hasEmailP = false;
            items.forEach(el => {
                if(el.querySelector('.check-principal').value === "true" && el.getAttribute('data-cat') === 'CORREO') hasEmailP = true;
            });

            if (!hasEmailP) return Swal.fire({ icon: 'error', title: 'Dato Requerido', text: 'Debe marcar un Correo como PRINCIPAL.' });

            const btnS = document.getElementById('btnSiguiente');
            btnS.disabled = true;
            btnS.innerHTML = '<span class="spinner-border spinner-border-sm me-2"></span> PROCESANDO...';

            fetch("{% url 'guardar_paso2_v2' %}", {
                method: "POST",
                body: new FormData(this),
                headers: { "X-CSRFToken": "{{ csrf_token }}" }
            })
            .then(r => r.json())
            .then(data => {
                if (data.success) {
                    Swal.fire({ icon: 'success', title: '¡Fino!', text: 'Información actualizada.', timer: 1500, showConfirmButton: false })
                    .then(() => { window.location.href = "{% url 'paso3_laboral_f1' %}"; });
                } else {
                    Swal.fire({ icon: 'error', title: 'Falla', text: data.error });
                    btnS.disabled = false; btnS.innerHTML = 'GUARDAR Y CONTINUAR <i class="fas fa-arrow-right ms-2"></i>';
                }
            })
            .catch(() => {
                Swal.fire({ icon: 'error', title: 'Error', text: 'Falla de conexión con el servidor.' });
                btnS.disabled = false; btnS.innerHTML = 'GUARDAR Y CONTINUAR <i class="fas fa-arrow-right ms-2"></i>';
            });
        });

        // --- [5. EVENTOS DELEGADOS (Para los botones remove de carga inicial)] ---
        lista.addEventListener('click', function(e) {
            if (e.target.closest('.btn-remove')) {
                const item = e.target.closest('.item-contacto');
                const isP = item.querySelector('.check-principal').value === "true";
                item.remove();
                if(isP) Swal.fire({ toast: true, position: 'top-end', icon: 'info', title: 'Asigne un nuevo principal', showConfirmButton: false, timer: 3000 });
                if (lista.children.length === 0) lista.innerHTML = '<div class="col-12 text-center py-4 text-muted" id="msgVacio"><i class="fas fa-info-circle me-1"></i> Lista vacía.</div>';
            }
        });

        // --- [6. VOLVER CON CONFIRMACIÓN] ---
        btnVolver.onclick = function(e) {
            if (lista.querySelectorAll('.item-contacto').length > 0) {
                e.preventDefault();
                Swal.fire({ title: '¿Deseas volver?', text: 'Los cambios no guardados se perderán.', icon: 'warning', showCancelButton: true, confirmButtonText: 'Sí, volver' })
                .then(r => { if(r.isConfirmed) window.location.href = this.href; });
            }
        };
    });

<script>
document.addEventListener('DOMContentLoaded', function() {
    
    // --- [1. DEFINICIÓN DE REFERENCIAS Y CONSTANTES] ---
    const form = document.getElementById('formPaso3');
    const cargoEst = document.getElementById('cargo_est_id');
    const cargoGen = document.getElementById('cargo_gen_id');
    const sede = document.getElementById('sede_id');
    const nivel = document.getElementById('nivel_id');
    const unidadN1 = document.getElementById('unidad_n1');
    const subContainer = document.getElementById('container-subunidades');
    const regimen = document.getElementById('select_regimen');
    const nomina = document.getElementById('codigo_nomina');
    const btnSiguiente = document.getElementById('btnSiguiente');

    /**
     * [2. REGLA DE NEGOCIO: LÍMITE DE VISUALIZACIÓN AJAX]
     * Define qué niveles podrá ver el usuario según el cargo seleccionado.
     */
    const getAJAXLimit = () => {
        const text = cargoEst.options[cargoEst.selectedIndex]?.text.toUpperCase() || "";
        if (text.includes("GERENTE")) return 1; // Solo ve Nivel 1
        if (text.includes("JEFE DE DIVISIÓN")) return 2; // Solo ve hasta Nivel 2
        return 3; // Analistas, Pasantes y Coordinadores ven hasta Nivel 3
    };

    /**
     * [3. REGLA DE NEGOCIO: PROFUNDIDAD MÍNIMA OBLIGATORIA]
     * Define qué niveles son OBLIGATORIOS de llenar para poder guardar.
     */
    const getMinRequiredLevel = () => {
        const text = cargoEst.options[cargoEst.selectedIndex]?.text.toUpperCase() || "";
        if (text.includes("GERENTE")) return 1;
        if (text.includes("JEFE DE DIVISIÓN")) return 2;
        if (text.includes("COORDINADOR")) return 3;
        return 1; // Pasantes y Analistas: Obligatorio solo Unidad Organizativa (Nivel 1)
    };

    /**
     * [4. ACTIVACIÓN POR CARGO ESTRUCTURAL]
     * Desbloquea la ubicación y autocompleta el genérico.
     */
    cargoEst.addEventListener('change', function() {
        const genId = this.options[this.selectedIndex].getAttribute('data-generico');
        if (genId) { 
            cargoGen.value = genId; 
            cargoGen.classList.add('is-valid'); 
        }
        
        // Desbloqueo láser de ubicación
        sede.disabled = false; 
        nivel.disabled = false;
        
        // Limpieza de subunidades si el usuario cambia el cargo en pleno proceso
        subContainer.innerHTML = '';
        if (sede.value && nivel.value) cargarUnidadesAJAX();
    });

    /**
     * [5. MOTOR DE CARGA DINÁMICA (AJAX)]
     * Inyecta elementos SELECT al DOM basándose en la jerarquía y reglas de negocio.
     */
    const cargarUnidadesAJAX = (superiorId = null, targetLevel = 1) => {
        // RESPETO AL LÍMITE DE VISUALIZACIÓN POR CARGO
        if (targetLevel > getAJAXLimit()) return;

        const sedeId = sede.value;
        const nivelId = nivel.value;
        if (!sedeId || !nivelId) return;

        let url = `{% url 'get_unidades_v2' %}?sede_id=${sedeId}&nivel_id=${nivelId}`;
        if (superiorId) url += `&superior_id=${superiorId}`;

        fetch(url).then(r => r.json()).then(data => {
            if (data.length > 0) {
                if (!superiorId) {
                    // LLENADO DE UNIDAD ORGANIZATIVA (NIVEL 1)
                    unidadN1.disabled = false;
                    unidadN1.innerHTML = '<option value="" selected disabled>...</option>';
                    data.forEach(u => {
                        unidadN1.innerHTML += `<option value="${u.id}">${u.nombre_unidad.toUpperCase()}</option>`;
                    });
                } else {
                    // FABRICACIÓN DINÁMICA DE SUBUNIDADES (NIVEL 2 Y 3)
                    const colDiv = document.createElement('div');
                    // Responsividad alineada al Paso 1
                    colDiv.className = "col-12 col-md-6 col-lg-4 fade-in-combo mt-2";
                    
                    const labelStr = targetLevel === 2 ? "Dependencia Adscrita" : "Área Operativa";
                    
                    // Solo es requerido si el cargo lo exige por jerarquía
                    const isMandatory = targetLevel <= getMinRequiredLevel();
                    const star = isMandatory ? ' <span class="req-pin">*</span>' : '';

                    colDiv.innerHTML = `
                        <label class="form-label">${labelStr}${star}</label>
                        <select class="form-select custom-input select-jerarquia" data-next="${targetLevel + 1}" ${isMandatory ? 'required' : ''}>
                            <option value="" selected disabled>...</option>
                            ${data.map(u => `<option value="${u.id}">${u.nombre_unidad.toUpperCase()}</option>`).join('')}
                        </select>

    (function() {

        const parentescosUnicos = ["MADRE", "PADRE", "ESPOSO", "ESPOSA", "CONYUGE"];

        function obtenerEdad(fechaString) {
            const hoy = new Date();
            const cumple = new Date(fechaString);
            let edad = hoy.getFullYear() - cumple.getFullYear();
            const m = hoy.getMonth() - cumple.getMonth();
            if (m < 0 || (m === 0 && hoy.getDate() < cumple.getDate())) { edad--; }
            return edad;
        }

        inpNacimiento.onchange = function() {
            if (this.value) {
                const edad = obtenerEdad(this.value);
                if (edad >= 10) {
                    reqCedula.style.display = 'inline';
                    inpCedula.placeholder = "V-00000000 (REQUERIDA)";
                } else {
                    reqCedula.style.display = 'none';
                    inpCedula.placeholder = "V-00000000 (OPCIONAL)";
                }
            }
        };

        document.getElementById('btnAddFamiliar').onclick = function() {
            const cedeVal = inpCedula.value.trim();

            const a2 = document.getElementById('temp_apellido2').value.trim().toUpperCase();

                return;
            }

            const edadActual = obtenerEdad(naciVal);
            if (edadActual >= 10 && !cedeVal) {
                inpCedula.classList.add('is-invalid');
                Swal.fire({ icon: 'error', title: 'Cédula Obligatoria', text: `A partir de los 10 años la cédula es requerida.`, background: swalBg, color: swalColor });
                return;
            }

            const parentescoTxt = parentesco.options[parentesco.selectedIndex].text.toUpperCase();

            if (parentescosUnicos.includes(parentescoTxt)) {
                const yaExiste = Array.from(listaF.querySelectorAll('.badge-parentesco')).some(el => el.innerText.toUpperCase() === parentescoTxt);
                if (yaExiste) {
                    Swal.fire({ icon: 'error', title: 'Registro Duplicado', text: `Ya registró a un familiar como ${parentescoTxt}.`, background: swalBg, color: swalColor });
                    return;
                }
            }

            const col = document.createElement('div');

                            <span class="badge bg-light text-dark border ms-1">${edadActual} AÑOS</span>
